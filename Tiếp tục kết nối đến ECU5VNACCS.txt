USE ECUS5VNACCS;
GO
IF OBJECT_ID('dbo.fn_TK_TotalItems_v2','FN') IS NOT NULL DROP FUNCTION dbo.fn_TK_TotalItems_v2;
GO
CREATE FUNCTION dbo.fn_TK_TotalItems_v2 (@SOTK NVARCHAR(50))
RETURNS INT
AS
BEGIN
  DECLARE @n INT;

  -- Fallback an toàn: đếm theo SOTK ở bảng chi tiết
  SELECT @n = COUNT(*)
  FROM dbo.DHANGMDDK
  WHERE CAST(SOTK AS NVARCHAR(50)) = @SOTK;

  RETURN ISNULL(@n,0);
END
GO

USE ECUS5VNACCS;
GO
/* Đếm số dòng C/O (giữ nguyên) */
IF OBJECT_ID('dbo.fn_TK_CO_Count','FN') IS NOT NULL DROP FUNCTION dbo.fn_TK_CO_Count;
GO
CREATE FUNCTION dbo.fn_TK_CO_Count(@SOTK NVARCHAR(50))
RETURNS INT
AS
BEGIN
  DECLARE @n INT;

  SELECT @n = SUM(
             CASE 
               WHEN LEFT(UPPER(LTRIM(RTRIM(CAST(TS_XNK_MA_BT AS NVARCHAR(10))))),3) LIKE N'B__'
                    AND LEFT(UPPER(LTRIM(RTRIM(CAST(TS_XNK_MA_BT AS NVARCHAR(10))))),3) 
                        NOT IN (N'B01', N'B02', N'B03', N'B30')
                 THEN 1
               ELSE 0
             END)
  FROM dbo.DHANGMDDK
  WHERE CAST(SOTK AS NVARCHAR(50)) = @SOTK;

  RETURN ISNULL(@n, 0);
END
GO

/* (MỚI) Trả về số (kiểu NVARCHAR) hoặc rỗng nếu = 0 */
IF OBJECT_ID('dbo.fn_TK_CO_CountStr','FN') IS NOT NULL DROP FUNCTION dbo.fn_TK_CO_CountStr;
GO
CREATE FUNCTION dbo.fn_TK_CO_CountStr(@SOTK NVARCHAR(50))
RETURNS NVARCHAR(10)
AS
BEGIN
  DECLARE @n INT = dbo.fn_TK_CO_Count(@SOTK);

  -- Nếu không có C/O thì trả rỗng (ECUS sẽ không hiển thị gì)
  IF @n IS NULL OR @n = 0 RETURN N'';

  -- Có C/O: trả riêng con số
  RETURN CAST(@n AS NVARCHAR(10));
END
GO

/* (TUỲ CHỌN) Nếu bạn muốn cột số trống khi 0, dùng hàm này */
IF OBJECT_ID('dbo.fn_TK_CO_CountOrNull','FN') IS NOT NULL DROP FUNCTION dbo.fn_TK_CO_CountOrNull;
GO
CREATE FUNCTION dbo.fn_TK_CO_CountOrNull(@SOTK NVARCHAR(50))
RETURNS INT
AS
BEGIN
  DECLARE @n INT = dbo.fn_TK_CO_Count(@SOTK);
  IF @n IS NULL OR @n = 0 RETURN NULL;  -- sẽ hiển thị trống
  RETURN @n;
END
GO
